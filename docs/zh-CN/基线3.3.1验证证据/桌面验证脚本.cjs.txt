const {_electron}=require('playwright');
const {expect}=require('@playwright/test');
const fs=require('fs');const path=require('path');const assert=require('node:assert/strict');
const root=process.cwd();const dir=path.join(root,'scratch/phase25/artifacts');
const userDir=path.join(root,'scratch/phase25/native-profile-word-boundary');
const executable=path.join(root,'scratch/phase25/delivery-check/Discourse Analysis.app/Contents/MacOS/Discourse Analysis');
const result={testNotes:['Initial custom smoke split inside beginning, then the original merge inserted a separator space by design; use a word boundary here. Original upstream tests unchanged.'],started:new Date().toISOString(),executable,steps:[],pageErrors:[],networkFailures:[]};
const persist=()=>fs.writeFileSync(path.join(dir,'native-smoke.json'),JSON.stringify(result,null,2));
async function step(name,fn){try{const detail=await fn();result.steps.push({name,status:'pass',detail:detail??null});console.log('PASS',name);persist();return true;}catch(e){result.steps.push({name,status:'fail',error:String(e.message)});console.log('FAIL',name,String(e.message).slice(0,350));persist();return false;}}
let app,page;
async function start(){
 app=await _electron.launch({executablePath:executable,args:['--user-data-dir='+userDir],cwd:root,artifactsDir:dir,acceptDownloads:true,timeout:30000});
 const info=await app.evaluate(({app})=>({version:app.getVersion(),packaged:app.isPackaged,userData:app.getPath('userData'),sessionData:app.getPath('sessionData'),arch:process.arch,electron:process.versions.electron}));
 assert.equal(info.userData,userDir);assert.equal(info.sessionData,userDir);assert.equal(info.version,'3.3.1');assert.equal(info.packaged,true);assert.equal(info.arch,'arm64');
 result.environment=info;persist();page=await app.firstWindow();page.setDefaultTimeout(15000);result.initialUrl=page.url();persist();
 page.on('pageerror',e=>{result.pageErrors.push(String(e.message));persist();});
 page.on('requestfailed',r=>{try{result.networkFailures.push({host:new URL(r.url()).hostname,error:r.failure()?.errorText});}catch{};persist();});
 await page.screenshot({path:path.join(dir,'00-startup.png')});await page.locator('#fetchBtn').waitFor({state:'visible',timeout:90000});
 return info;
}
async function chooseProfile(id){await page.locator('#projectSettingsBtn').click();await page.locator('#profileSelect').selectOption(id);await page.locator('#closeSettingsBtn').click();await expect(page.locator('#settingsModal')).toBeHidden();assert.equal(await page.evaluate(()=>DA_PROFILES.getActive().id),id);}
async function download(action,name){await page.locator('#exportMenuBtn').click();const pending=page.waitForEvent('download',{timeout:45000});await page.locator(`#exportMenu [data-action="${action}"]`).click();const d=await pending;const target=path.join(dir,name);await d.saveAs(target);assert.equal(await d.failure(),null);return target;}
async function openFile(target){await page.locator('#openMenuBtn').click();const pending=page.waitForEvent('filechooser');await page.getByRole('button',{name:'📂 Open local file',exact:true}).click();await (await pending).setFiles(target);await expect(page.locator('.proposition-block')).toHaveCount(3);await expect(page.locator('.bracket-group')).toHaveCount(1);assert.equal(await page.evaluate(()=>DA_STATE.brackets[0].type),'action-purpose');}
(async()=>{
 try{
  const started=await step('ZIP 解压后的 arm64 应用启动及数据隔离',start);if(!started)return;
  await step('首次启动欢迎页',async()=>{await expect(page.locator('#welcomeModal')).toBeVisible();await page.locator('#welcomeName').fill('Baseline Smoke Test');await page.locator('#welcomeSaveBtn').click();await expect(page.locator('#welcomeModal')).toBeHidden();return 'Test identity saved only in isolated profile';});
  const fetched=await step('通过真实网络载入 John 1:1-3（无响应替身）',async()=>{await page.locator('#versionSelect').selectOption('esv');await page.locator('#passageInput').fill('John 1:1-3');await page.locator('#fetchBtn').click();await expect(page.locator('.proposition-block')).toHaveCount(3,{timeout:45000});const state=await page.evaluate(()=>({ref:DA_STATE.passageRef,verses:DA_STATE.verseRefs,texts:DA_STATE.propositions,source:document.getElementById('copyrightLabel').textContent}));assert.ok(state.texts.every(t=>t.length>5));result.loadedText=state.texts;await page.screenshot({path:path.join(dir,'01-loaded.png')});return {ref:state.ref,verses:state.verses,source:state.source};});
  if(!fetched){await step('离线粘贴载入（仅为继续检查其他功能）',async()=>{await page.getByText('Or paste text to bracket or import',{exact:true}).click();await page.locator('#pasteText').fill('[1] In the beginning was the Word [2] The Word was with God [3] All things were made through him');await page.locator('#importPassageRef').fill('John 1:1-3');await page.locator('#importBtn').click();await expect(page.locator('.proposition-block')).toHaveCount(3);result.loadedText=await page.evaluate(()=>DA_STATE.propositions.slice());});}
  await step('真实键盘拆分／合并并保持文字',async()=>{const before=await page.evaluate(()=>DA_STATE.propositions.slice());const caret=async(i,offset)=>page.evaluate(({i,offset})=>{const span=document.querySelectorAll('.proposition-block .proposition-text')[i];span.focus();const range=document.createRange();range.setStart(span.firstChild,offset);range.collapse(true);const sel=window.getSelection();sel.removeAllRanges();sel.addRange(range);},{i,offset});const boundary=before[0].indexOf(' ',8);assert.ok(boundary>0);await caret(0,boundary);await page.keyboard.press('Enter');await expect(page.locator('.proposition-block')).toHaveCount(4);await caret(1,0);await page.keyboard.press('Backspace');await expect(page.locator('.proposition-block')).toHaveCount(3);assert.deepEqual(await page.evaluate(()=>DA_STATE.propositions.slice()),before);});
  await step('用圆点与选择器建立 Action-Purpose 关系',async()=>{await page.locator('.prop-dot').nth(0).click();await page.locator('.prop-dot').nth(1).click();await expect(page.locator('#labelPicker')).toBeVisible();await page.locator('#labelPicker button.action-purpose').click();await expect(page.locator('.bracket-group.action-purpose')).toHaveCount(1);assert.equal(await page.evaluate(()=>DA_STATE.brackets[0].type),'action-purpose');});
  await step('五套 profile 切换及差异保持',async()=>{const records=[];for(const id of ['josiah','biblearc','gurtner','beale','schreiner']){await chooseProfile(id);const p=await page.evaluate(()=>({id:DA_PROFILES.getActive().id,name:DA_PROFILES.getName('action-purpose'),abbr:DA_PROFILES.getAbbr('action-purpose'),label:DA_RENDERER.getBracketLabels('action-purpose'),storedType:DA_STATE.brackets[0].type}));assert.equal(p.storedType,'action-purpose');if(['gurtner','beale'].includes(id))assert.match(p.name,/Means-End/);else assert.match(p.name,/Action-Purpose/);records.push(p);}await page.screenshot({path:path.join(dir,'02-profiles.png')});return records;});
  let saved;
  await step('保存项目 JSON',async()=>{const pending=page.waitForEvent('download');await page.locator('#saveBtn').click();const d=await pending;saved=path.join(dir,'baseline-project.json');await d.saveAs(saved);const j=JSON.parse(fs.readFileSync(saved,'utf8'));assert.equal(j.version,3);assert.equal(j.propositions.length,3);assert.equal(j.brackets[0].type,'action-purpose');assert.deepEqual(j.propositions,result.loadedText);return {filename:'baseline-project.json',version:j.version,brackets:j.brackets.length};});
  if(saved)await step('打开已保存的 JSON 并核对内容',async()=>{await openFile(saved);assert.deepEqual(await page.evaluate(()=>DA_STATE.propositions.slice()),result.loadedText);});
  let png,pdf;
  await step('导出 PNG 图片及项目数据',async()=>{png=await download('png','baseline-export.png');const b=fs.readFileSync(png);assert.equal(b.subarray(1,4).toString(),'PNG');assert.ok(b.includes(Buffer.from('BibleBracket')));return {bytes:b.length};});
  if(png)await step('通过打开文件重新导入 PNG',async()=>{await openFile(png);assert.deepEqual(await page.evaluate(()=>DA_STATE.propositions.slice()),result.loadedText);});
  await step('导出 PDF',async()=>{pdf=await download('pdf','baseline-export.pdf');const b=fs.readFileSync(pdf);assert.equal(b.subarray(0,5).toString(),'%PDF-');assert.ok(b.includes(Buffer.from('BibleBracketDNA:')));return {bytes:b.length};});
  if(pdf)await step('通过打开文件重新导入 PDF',async()=>{await openFile(pdf);assert.deepEqual(await page.evaluate(()=>DA_STATE.propositions.slice()),result.loadedText);});
  await step('最终界面和浏览器错误检查',async()=>{await page.screenshot({path:path.join(dir,'03-final.png')});assert.deepEqual(result.pageErrors,[]);return {title:await page.title(),bodyLanguage:await page.locator('html').getAttribute('lang')};});
  await step('关闭后重新启动，记号方案保留',async()=>{await app.close();app=null;const info=await start();assert.equal(await page.evaluate(()=>DA_PROFILES.getActive().id),'schreiner');await expect(page.locator('#welcomeModal')).toBeHidden();return info;});
 }catch(e){result.fatal=String(e.stack||e);console.log('FATAL',result.fatal);}
 finally{if(app)await app.close().catch(()=>{});result.finished=new Date().toISOString();result.passed=result.steps.filter(s=>s.status==='pass').length;result.failed=result.steps.filter(s=>s.status==='fail').length;persist();console.log(JSON.stringify({passed:result.passed,failed:result.failed,fatal:result.fatal||null}));if(result.failed||result.fatal)process.exitCode=1;}
})();
